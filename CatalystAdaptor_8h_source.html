56struct Button {
57 Button() = default;
58
59 // explicit not explicit allows: if(my_btn) to work
60 operator bool() const { return value_m; }
61
62 // explicit Button(bool v) : value_m(v) {}
63 explicit Button(bool v){
64
65 if(v){ // Button is initialized pressd
66 value_m = true;
67 priorState_m=false;
68 }
69 else /* if(!v) */{ // Button is initialized unpressd
70 value_m=false;
71 priorState_m = false;
72 }
73
74 }
75
76
77 // Assignment operator: Button = bool
78 Button& operator=(bool v) {
79 if(v) { // Button is being pressed
80 if(!value_m && !priorState_m) { // True unpressed state
81 value_m = true;
82 priorState_m = false; // Fixed typo: was "priot_state"
83 }
84 else if(value_m && !priorState_m) { // Button was pressed last iteration and is still "being pushed down"
85 // Internal button needs to snap back! And mark it as "freshly snapped"
86 value_m = false;
87 priorState_m = true;
88 }
89 else if(!value_m && priorState_m) { // Button was pressed at some previous iteration and is still "being pushed down"
90 // Button needs to stay in freshly snapped back state
91 value_m = false;
92 priorState_m = true;
93 }
94 else if(value_m && priorState_m) { // Impossible state
95 throw IpplException("CatalystAdaptor::Button Assignment", "Impossible State: Button is malfunctioning!!!");
96 }
97 }
98 else { // Button is unpressed
99 value_m = false;
100 priorState_m = false;
101 }
102 return *this; // CRITICAL: Return reference to this object for chaining
103 }
104
105 // Friend function to overload << operator for output streams
106 friend std::ostream& operator<<(std::ostream& os, const Button& btn) {
107 os << (btn.value_m ? "PUSHED" : "not PUSHED");
108 return os;
109 }
110
111 private:
112 bool value_m = false;
113 bool priorState_m = false;
114};// Button

137struct GhostKeyHash {
138 std::size_t operator()(const GhostKey_t& k) const {
139 // Get the hash for each element in the tuple
140 auto h1 = std::hash<const void*>{}(std::get<0>(k));
141 auto h2 = std::hash<const void*>{}(std::get<1>(k));
142 auto h3 = std::hash<size_t>{}(std::get<2>(k));
143
144 // Combine the hashes. This is a common pattern (based on boost::hash_combine)
145 // It xors and bit-shifts to mix the bits well.
146 h1 ^= h2 + 0x9e3779b9 + (h1 << 6) + (h1 >> 2);
147 h1 ^= h3 + 0x9e3779b9 + (h1 << 6) + (h1 >> 2);
148 return h1;
149 }
