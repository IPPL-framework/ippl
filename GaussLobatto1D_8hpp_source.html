28 template <typename Scalar>
29 void legendreAndDeriv(std::size_t n, Scalar x, Scalar& p, Scalar& dp) {
30 if (n == 0) {
31 p = Scalar(1);
32 dp = Scalar(0);
33 return;
34 }
35 if (n == 1) {
36 p = x;
37 dp = Scalar(1);
38 return;
39 }
40 Scalar p0 = Scalar(1);
41 Scalar p1 = x;
42 for (std::size_t k = 2; k <= n; ++k) {
43 const Scalar pk =
44 ((2 * static_cast<Scalar>(k) - 1) * x * p1 - (static_cast<Scalar>(k) - 1) * p0)
45 / static_cast<Scalar>(k);
46 p0 = p1;
47 p1 = pk;
48 }
49 p = p1;
50 const Scalar pm1 = p0;
51 dp = static_cast<Scalar>(n) * (x * p - pm1) / (x * x - Scalar(1));
52 }

58 template <typename Scalar>
59 void fillGLLWeights(std::size_t n, Scalar* nodes, Scalar* weights) {
60 const Scalar boundary = Scalar(2) / (static_cast<Scalar>(n) * static_cast<Scalar>(n - 1));
61 weights[0] = boundary;
62 weights[n - 1] = boundary;
63 const Scalar denomFactor = static_cast<Scalar>(n) * static_cast<Scalar>(n - 1);
64 for (std::size_t i = 1; i < n - 1; ++i) {
65 const Scalar p = evalLegendre(n - 1, nodes[i]);
66 weights[i] = Scalar(2) / (denomFactor * p * p);
67 }
68 }

74 template <typename Scalar>
75 Scalar brentLegendreDerivRoot(std::size_t deg, Scalar lo, Scalar hi, std::size_t maxIter = 200) {
76 // Find root of P'_deg on (lo, hi).
77 auto f = [&](Scalar x) {
78 Scalar p, dp;
79 legendreAndDeriv(deg, x, p, dp);
80 return dp;
81 };
82 Scalar a = lo;
83 Scalar b = hi;
84 Scalar fa = f(a);
85 Scalar fb = f(b);
86 if (fa * fb > Scalar(0)) {
87 throw std::runtime_error("nodes1d::brentLegendreDerivRoot: no sign change");
88 }
89 if (Kokkos::abs(fa) < Kokkos::abs(fb)) {
90 std::swap(a, b);
91 std::swap(fa, fb);
92 }
93 Scalar c = a;
94 Scalar fc = fa;
95 bool mflag = true;
96 Scalar d = Scalar(0);
97 for (std::size_t it = 0; it < maxIter; ++it) {
98 if (fb == Scalar(0) || Kokkos::abs(b - a) < Scalar(2e-16)) {
99 return b;
100 }
101 Scalar s;
102 if (fa != fc && fb != fc) {
103 s = a * fb * fc / ((fa - fb) * (fa - fc)) + b * fa * fc / ((fb - fa) * (fb - fc))
104 + c * fa * fb / ((fc - fa) * (fc - fb));
105 } else {
106 s = b - fb * (b - a) / (fb - fa);
107 }
108 const bool cond1 = (s < (Scalar(3) * a + b) / Scalar(4) || s > b);
109 const bool cond2 = mflag && Kokkos::abs(s - b) >= Kokkos::abs(b - c) / Scalar(2);
110 const bool cond3 = !mflag && Kokkos::abs(s - b) >= Kokkos::abs(c - d) / Scalar(2);
111 const bool cond4 = mflag && Kokkos::abs(b - c) < Scalar(2e-16);
112 const bool cond5 = !mflag && Kokkos::abs(c - d) < Scalar(2e-16);
113 if (cond1 || cond2 || cond3 || cond4 || cond5) {
114 s = (a + b) / Scalar(2);
115 mflag = true;
116 } else {
117 mflag = false;
118 }
119 const Scalar fs = f(s);
120 d = c;
121 c = b;
122 fc = fb;
123 if (fa * fs < Scalar(0)) {
124 b = s;
125 fb = fs;
126 } else {
127 a = s;
128 fa = fs;
129 }
130 if (Kokkos::abs(fa) < Kokkos::abs(fb)) {
131 std::swap(a, b);
132 std::swap(fa, fb);
133 }
134 }
135 return b;
136 }

182 template <typename Scalar>
183 void computeGaussLobattoNewton(std::size_t n, Scalar* nodes, Scalar* weights,
184 std::size_t maxNewtonIterations,
185 std::size_t minNewtonIterations) {
186 nodes[0] = Scalar(-1);
187 nodes[n - 1] = Scalar(1);
188
189 if (n == 2) {
190 weights[0] = Scalar(1);
191 weights[1] = Scalar(1);
192 return;
193 }
194
195 const std::size_t deg = n - 1;
196 const Scalar tol = Scalar(2e-16);
197 const Scalar sepTol = Scalar(1e-13);
198 const std::size_t maxIts = (maxNewtonIterations < 40) ? 40 : maxNewtonIterations;
199 const std::size_t nInt = n - 2;
200
201 // Interior index j = 0..nInt-1 maps to ascending nodes[1..n-2].
202 for (std::size_t j = 0; j < nInt; ++j) {
203 const Scalar lo =
204 (j == 0) ? Scalar(-1) + Scalar(1e-14) : nodes[j] + sepTol;
205 Scalar hi = Scalar(1) - Scalar(1e-14);
206 if (j + 1 < nInt) {
207 const Scalar nextAsym = asymptoticJacobiRoot(j + 1, nInt, Scalar(1), Scalar(1));
208 const Scalar curAsym = asymptoticJacobiRoot(j, nInt, Scalar(1), Scalar(1));
209 hi = Scalar(0.5) * (curAsym + nextAsym);
210 if (hi <= lo) {
211 hi = Scalar(1) - Scalar(1e-14);
212 }
213 }
214
215 Scalar guesses[2] = {
216 asymptoticJacobiRoot(j, nInt, Scalar(1), Scalar(1)),
217 -Kokkos::cos(Kokkos::numbers::pi_v<Scalar> * static_cast<Scalar>(j + 1)
218 / static_cast<Scalar>(deg)),
219 };
220
221 bool accepted = false;
222 Scalar z = Scalar(0);
223 for (Scalar guess : guesses) {
224 z = guess;
225 if (z <= lo || z >= hi) {
226 z = Scalar(0.5) * (lo + hi);
227 }
228 bool converged = false;
229 for (std::size_t it = 0; it < maxIts; ++it) {
230 Scalar p, dp;
231 legendreAndDeriv(deg, z, p, dp);
232 const Scalar denom = (Scalar(1) - z * z);
233 if (Kokkos::abs(denom) < Scalar(1e-30)) {
234 break;
235 }
236 const Scalar ddp =
237 (Scalar(2) * z * dp - static_cast<Scalar>(deg * (deg + 1)) * p) / denom;
238 const Scalar delta = dp / ddp;
239 z -= delta;
240 if (z < lo) {
241 z = lo;
242 }
243 if (z > hi) {
244 z = hi;
245 }
246 if (it >= minNewtonIterations && Kokkos::abs(delta) <= tol) {
247 converged = true;
248 break;
249 }
250 }
251 if (!converged || z <= lo || z >= hi || isDuplicateAmong(nodes + 1, j, z, sepTol)) {
252 continue;
253 }
254 accepted = true;
255 break;
256 }
257
258 if (!accepted) {
259 // Sample P' for a sign change on (lo, hi), then Brent.
260 auto fprime = [&](Scalar x) {
261 Scalar p, dp;
262 legendreAndDeriv(deg, x, p, dp);
263 return dp;
264 };
265 Scalar blo = lo;
266 Scalar bhi = hi;
267 Scalar fprev = fprime(blo);
268 bool found = false;
269 for (int s = 1; s <= 128; ++s) {
270 const Scalar x =
271 lo + (hi - lo) * (static_cast<Scalar>(s) / Scalar(128));
272 const Scalar fx = fprime(x);
273 if (fprev * fx <= Scalar(0)) {
274 blo = lo + (hi - lo) * (static_cast<Scalar>(s - 1) / Scalar(128));
275 bhi = x;
276 found = true;
277 break;
278 }
279 fprev = fx;
280 }
281 if (!found) {
282 blo = lo;
283 bhi = Scalar(1) - Scalar(1e-14);
284 fprev = fprime(blo);
285 for (int s = 1; s <= 256; ++s) {
286 const Scalar x =
287 blo + (bhi - blo) * (static_cast<Scalar>(s) / Scalar(256));
288 const Scalar fx = fprime(x);
289 if (fprev * fx <= Scalar(0)) {
290 const Scalar left =
291 blo + (bhi - blo) * (static_cast<Scalar>(s - 1) / Scalar(256));
292 blo = left;
293 bhi = x;
294 found = true;
295 break;
296 }
297 fprev = fx;
298 }
299 }
300 if (!found) {
301 throw std::runtime_error(
302 "nodes1d::computeGaussLobatto (Newton): could not bracket interior j="
303 + std::to_string(j) + ", n=" + std::to_string(n));
304 }
305 z = brentLegendreDerivRoot(deg, blo, bhi);
306 if (z <= lo || isDuplicateAmong(nodes + 1, j, z, sepTol)) {
307 throwDuplicateNodes("nodes1d::computeGaussLobatto (Newton)", n, j + 1, j,
308 static_cast<double>(z),
309 static_cast<double>(nodes[j]));
310 }
311 }
312 nodes[j + 1] = z;
313 }
314
315 assertDistinctSortedNodes("nodes1d::computeGaussLobatto (Newton)", nodes, n, sepTol);
316 fillGLLWeights(n, nodes, weights);
317 }
