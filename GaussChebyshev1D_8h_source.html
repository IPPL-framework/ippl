41 template <typename Scalar = double>
42 KOKKOS_INLINE_FUNCTION Scalar chebyshevNode(std::size_t i, std::size_t numNodes) {
43 return -Kokkos::cos((2.0 * static_cast<Scalar>(i) + 1.0) * Kokkos::numbers::pi_v<Scalar>

59 template <typename Scalar = double>
60 void computeGaussChebyshev(std::size_t n, Scalar* nodes, Scalar* weights) {
61 assert(n >= 1);
62 assert(nodes != nullptr && weights != nullptr);
63 const Scalar w = Kokkos::numbers::pi_v<Scalar> / static_cast<Scalar>(n);
64 for (std::size_t i = 0; i < n; ++i) {
65 nodes[i] = chebyshevNode<Scalar>(i, n);
66 weights[i] = w;
67 }
68 }
