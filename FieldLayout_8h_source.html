108 template <unsigned Dim>
109 unsigned int indexToFace(unsigned int index) {
110 // facets are group low/high by axis
111 unsigned int axis = index / 2;
112 // the digit to subtract is determined by whether the index describes an upper
113 // face (even index) or lower face (odd index) and that digit's position is
114 // determined by the axis of the face
115 unsigned int toRemove = (2 - index % 2) * countHypercubes(axis);
116 // start with all 2s (in base 3) and change the correct digit to get the encoded face
117 return countHypercubes(Dim) - 1 - toRemove;
118 }

127 template <
128 unsigned Dim, typename... CubeTags,
129 typename = std::enable_if_t<sizeof...(CubeTags) == Dim - 1>,
130 typename = std::enable_if_t<std::conjunction_v<std::is_same<e_cube_tag, CubeTags>...>>>
131 unsigned int getCube(e_cube_tag tag, CubeTags... tags) {
132 if constexpr (Dim == 1) {
133 return tag;
134 } else {
135 return tag + 3 * getCube<Dim - 1>(tags...);
136 }
137 }
