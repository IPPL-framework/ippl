36 template <typename Scalar = double>
37 KOKKOS_INLINE_FUNCTION Scalar evalLegendre(std::size_t degree, Scalar x) {
38 if (degree == 0) {
39 return Scalar(1);
40 }
41 if (degree == 1) {
42 return x;
43 }
44 Scalar p0 = Scalar(1);
45 Scalar p1 = x;
46 for (std::size_t k = 2; k <= degree; ++k) {
47 const Scalar pk =
48 ((2 * static_cast<Scalar>(k) - 1) * x * p1 - (static_cast<Scalar>(k) - 1) * p0)
49 / static_cast<Scalar>(k);
50 p0 = p1;
51 p1 = pk;
52 }
53 return p1;
54 }
