66 template <typename Scalar>
67 Scalar jacobiCompanionDiagonal(std::size_t k, Scalar alpha, Scalar beta) {
68 const Scalar ab = alpha + beta;
69 if (k == 0) {
70 return (beta - alpha) / (Scalar(2) + ab);
71 }
72 if (ab == Scalar(0)) {
73 return Scalar(0);
74 }
75 return (beta * beta - alpha * alpha)
76 / ((Scalar(2) * k + ab) * (Scalar(2) * k + ab + Scalar(2)));
77 }

81 Scalar jacobiCompanionSubdiag(std::size_t k, Scalar alpha, Scalar beta) {
82 const Scalar ab = alpha + beta;
83 const Scalar kk = static_cast<Scalar>(k);

108 template <typename Scalar>
109 void evalJacobiAndDeriv(std::size_t n, Scalar alpha, Scalar beta, Scalar x, Scalar& p,
110 Scalar& dp) {
111 if (n == 0) {
112 p = Scalar(1);
113 dp = Scalar(0);
114 return;
115 }
116 const Scalar alfbet = alpha + beta;
117 Scalar p0 = Scalar(1);
118 Scalar p1 = (alpha - beta + (Scalar(2) + alfbet) * x) / Scalar(2);
119 if (n == 1) {
120 p = p1;
121 dp = (Scalar(2) + alfbet) / Scalar(2);
122 return;
123 }
124 for (std::size_t j = 2; j <= n; ++j) {
125 const Scalar temp = Scalar(2) * static_cast<Scalar>(j) + alfbet;
126 const Scalar a =
127 Scalar(2) * static_cast<Scalar>(j) * (static_cast<Scalar>(j) + alfbet) * (temp - Scalar(2));
128 const Scalar b =
129 (temp - Scalar(1))
130 * (alpha * alpha - beta * beta + temp * (temp - Scalar(2)) * x);
131 const Scalar c = Scalar(2) * (static_cast<Scalar>(j) - Scalar(1) + alpha)
132 * (static_cast<Scalar>(j) - Scalar(1) + beta) * temp;
133 const Scalar pn = (b * p1 - c * p0) / a;
134 p0 = p1;
135 p1 = pn;
136 }
137 p = p1;
138 // P_n' = n/(2x) * ... use recurrence identity:
139 // (1-x^2) P' = n/2 * ((α-β-(2n+α+β)x) P_n + 2(n+α)(n+β)/(2n+α+β) P_{n-1} wait —
140 // SciPy: df = 0.5*(n+α+β+1)*P_{n-1}^{(α+1,β+1)}
141 // From Newton weight path:
142 const Scalar temp = Scalar(2) * static_cast<Scalar>(n) + alfbet;
143 dp = (static_cast<Scalar>(n) * (alpha - beta - temp * x) * p1
144 + Scalar(2) * (static_cast<Scalar>(n) + alpha) * (static_cast<Scalar>(n) + beta) * p0)
145 / (temp * (Scalar(1) - x * x));
146 }

153 template <typename Scalar>
154 void tridiagonalQL(std::vector<Scalar>& d, std::vector<Scalar>& e, std::vector<Scalar>& z,
155 std::size_t n) {
156 const Scalar eps = Scalar(2e-16);
157 for (std::size_t i = 1; i < n; ++i) {
158 e[i - 1] = e[i];
159 }
160 e[n - 1] = Scalar(0);
161
162 for (std::size_t l = 0; l < n; ++l) {
163 std::size_t iter = 0;
164 for (;;) {
165 std::size_t m = l;
166 for (; m < n - 1; ++m) {
167 const Scalar dd = Kokkos::abs(d[m]) + Kokkos::abs(d[m + 1]);
168 if (Kokkos::abs(e[m]) + dd == dd) {
169 break;
170 }
171 }
172 if (m == l) {
173 break;
174 }
175 if (++iter > 100 * n) {
176 throw std::runtime_error(
177 "nodes1d::tridiagonalQL: failed to converge (Golub–Welsch)");
178 }
179
180 Scalar g = (d[l + 1] - d[l]) / (Scalar(2) * e[l]);
181 Scalar r = Kokkos::sqrt(g * g + Scalar(1));
182 g = d[m] - d[l] + e[l] / (g + (g >= Scalar(0) ? r : -r));
183 Scalar s = Scalar(1);
184 Scalar c = Scalar(1);
185 Scalar p = Scalar(0);
186 for (std::size_t i = m - 1; i + 1 > l; --i) {
187 const Scalar f = s * e[i];
188 const Scalar b = c * e[i];
189 if (Kokkos::abs(f) >= Kokkos::abs(g)) {
190 c = g / f;
191 r = Kokkos::sqrt(c * c + Scalar(1));
192 e[i + 1] = f * r;
193 s = Scalar(1) / r;
194 c *= s;
195 } else {
196 s = f / g;
197 r = Kokkos::sqrt(s * s + Scalar(1));
198 e[i + 1] = g * r;
199 c = Scalar(1) / r;
200 s *= c;
201 }
202 g = d[i + 1] - p;
203 r = (d[i] - g) * s + Scalar(2) * c * b;
204 p = s * r;
205 d[i + 1] = g + p;
206 g = c * r - b;
207
208 for (std::size_t k = 0; k < n; ++k) {
209 const Scalar f2 = z[k * n + (i + 1)];
210 z[k * n + (i + 1)] = s * z[k * n + i] + c * f2;
211 z[k * n + i] = c * z[k * n + i] - s * f2;
212 }
213 if (i == l) {
214 break;
215 }
216 }
217 d[l] -= p;
218 e[l] = g;
219 e[m] = Scalar(0);
220 (void)eps;
221 }
222 }
223 }

247 template <typename Scalar>
248 void finalizeGolubWelschResults(std::size_t n, Scalar alpha, Scalar beta, Scalar* nodes,
249 Scalar* weights, const std::vector<Scalar>& evals,
250 const std::vector<Scalar>& evecs, const std::vector<std::size_t>& perm) {
251 const Scalar mu0 =
252 Kokkos::pow(Scalar(2), alpha + beta + Scalar(1))
253 * Kokkos::exp(Kokkos::lgamma(alpha + Scalar(1)) + Kokkos::lgamma(beta + Scalar(1))
254 - Kokkos::lgamma(alpha + beta + Scalar(2)));
255
256 for (std::size_t i = 0; i < n; ++i) {
257 const std::size_t j = perm[i];
258 Scalar x = evals[j];
259 polishJacobiRoot(n, alpha, beta, x);
260 nodes[i] = x;
261 const Scalar v0 = evecs[0 * n + j];
262 weights[i] = mu0 * v0 * v0;
263 }
264
265 Scalar wsum = Scalar(0);
266 for (std::size_t i = 0; i < n; ++i) {
267 wsum += weights[i];
268 }
269 if (wsum > Scalar(0)) {
270 const Scalar scale = mu0 / wsum;
271 for (std::size_t i = 0; i < n; ++i) {
272 weights[i] *= scale;
273 }
274 }
275
276 if (alpha == beta) {
277 for (std::size_t i = 0; i < n; ++i) {
278 const std::size_t j = n - 1 - i;
279 if (i > j) {
280 break;
281 }
282 const Scalar xsym = Scalar(0.5) * (nodes[i] - nodes[j]);
283 const Scalar wsym = Scalar(0.5) * (weights[i] + weights[j]);
284 nodes[i] = xsym;
285 nodes[j] = -xsym;
286 weights[i] = weights[j] = wsym;
287 }
288 }

305 template <typename Scalar>
306 void denseSymmetricEigenJacobi(std::vector<Scalar>& A, std::vector<Scalar>& V, std::size_t n,
307 std::size_t maxSweeps = 64) {
308 V.assign(n * n, Scalar(0));
309 for (std::size_t i = 0; i < n; ++i) {
310 V[i * n + i] = Scalar(1);
311 }
312
313 const Scalar tol = Scalar(2e-16);
314 for (std::size_t sweep = 0; sweep < maxSweeps; ++sweep) {
315 Scalar off = Scalar(0);
316 for (std::size_t i = 0; i < n; ++i) {
317 for (std::size_t j = i + 1; j < n; ++j) {
318 off += Kokkos::abs(A[i * n + j]);
319 }
320 }
321 if (off <= tol) {
322 return;
323 }
324
325 for (std::size_t p = 0; p < n; ++p) {
326 for (std::size_t q = p + 1; q < n; ++q) {
327 const Scalar apq = A[p * n + q];
328 if (Kokkos::abs(apq) <= tol) {
329 continue;
330 }
331 const Scalar app = A[p * n + p];
332 const Scalar aqq = A[q * n + q];
333 const Scalar tau = (aqq - app) / (Scalar(2) * apq);
334 const Scalar t =
335 (tau >= Scalar(0))
336 ? Scalar(1) / (tau + Kokkos::sqrt(Scalar(1) + tau * tau))
337 : Scalar(-1) / (-tau + Kokkos::sqrt(Scalar(1) + tau * tau));
338 const Scalar c = Scalar(1) / Kokkos::sqrt(Scalar(1) + t * t);
339 const Scalar s = t * c;
340
341 A[p * n + p] = app - t * apq;
342 A[q * n + q] = aqq + t * apq;
343 A[p * n + q] = Scalar(0);
344 A[q * n + p] = Scalar(0);
345
346 for (std::size_t r = 0; r < n; ++r) {
347 if (r == p || r == q) {
348 continue;
349 }
350 const Scalar arp = A[r * n + p];
351 const Scalar arq = A[r * n + q];
352 A[r * n + p] = A[p * n + r] = c * arp - s * arq;
353 A[r * n + q] = A[q * n + r] = s * arp + c * arq;
354 }
355
356 for (std::size_t r = 0; r < n; ++r) {
357 const Scalar vrp = V[r * n + p];
358 const Scalar vrq = V[r * n + q];
359 V[r * n + p] = c * vrp - s * vrq;
360 V[r * n + q] = s * vrp + c * vrq;
361 }
362 }
363 }
364 }

368 template <typename Scalar>
369 void computeGaussJacobiGolubWelschDense(std::size_t n, Scalar alpha, Scalar beta, Scalar* nodes,
370 Scalar* weights) {
371 if (n == 1) {
372 gaussJacobiMu0SingleNode(alpha, beta, nodes, weights);
373 return;
374 }
375
376 std::vector<Scalar> A(n * n, Scalar(0));
377 for (std::size_t i = 0; i < n; ++i) {
378 A[i * n + i] = jacobiCompanionDiagonal(i, alpha, beta);
379 }
380 for (std::size_t k = 1; k < n; ++k) {
381 const Scalar b = jacobiCompanionSubdiag(k, alpha, beta);
382 A[(k - 1) * n + k] = b;
383 A[k * n + (k - 1)] = b;
384 }
385
386 std::vector<Scalar> V;
387 denseSymmetricEigenJacobi(A, V, n);
388

390 for (std::size_t i = 0; i < n; ++i) {
391 evals[i] = A[i * n + i];
392 }
393 std::vector<std::size_t> perm(n);
394 for (std::size_t i = 0; i < n; ++i) {
395 perm[i] = i;
396 }
397 std::sort(perm.begin(), perm.end(),
398 [&](std::size_t i, std::size_t j) { return evals[i] < evals[j]; });
399
400 finalizeGolubWelschResults(n, alpha, beta, nodes, weights, evals, V, perm);
401 }

441 template <typename Scalar>
442 Scalar stroudSecrestInitialGuess(std::size_t i, std::size_t n, Scalar alpha, Scalar beta,
443 const std::vector<Scalar>& uniqueSoFar) {
444 // Original formulas scan descending from +1. Map ascending index i via previously
445 // accepted ascending nodes when available.
446 Scalar r1, r2, r3;
447 Scalar z = (i > 0 && uniqueSoFar.size() >= i) ? uniqueSoFar[i - 1] : Scalar(0);
448
449 if (i == 0) {
450 const Scalar an = alpha / static_cast<Scalar>(n);
451 const Scalar bn = beta / static_cast<Scalar>(n);
452 r1 = (1.0 + alpha) * (2.78 / (4.0 + n * n) + 0.768 * an / static_cast<Scalar>(n));
453 r2 = 1.0 + 1.48 * an + 0.96 * bn + 0.452 * an * an + 0.83 * an * bn;
454 z = -(1.0 - r1 / r2); // mirror the first (+1)-side guess to the -1 side
455 } else if (i == 1) {
456 if (uniqueSoFar.empty()) {
457 return asymptoticJacobiRoot(i, n, alpha, beta);
458 }
459 r1 = (4.1 + beta) / ((1.0 + beta) * (1.0 + 0.156 * beta));
460 r2 = 1.0 + 0.06 * (n - 8.0) * (1.0 + 0.12 * beta) / static_cast<Scalar>(n);
461 r3 = 1.0 + 0.012 * alpha * (1.0 + 0.25 * Kokkos::abs(beta)) / static_cast<Scalar>(n);
462 z += (1.0 + z) * r1 * r2 * r3 * Scalar(0.25);
463 if (z > Scalar(0.9)) {
464 z = asymptoticJacobiRoot(i, n, alpha, beta);
465 }
466 } else {
467 z = asymptoticJacobiRoot(i, n, alpha, beta);
468 }
469 return z;
470 }

476 template <typename Scalar>
477 bool newtonOneRootBounded(std::size_t n, Scalar alpha, Scalar beta, Scalar& z, Scalar lo,
478 Scalar hi, std::size_t maxNewtonIterations,
479 std::size_t minNewtonIterations, Scalar& p_out, Scalar& dp_out,
480 Scalar& pnm1_out, Scalar& temp_out) {
481 const Scalar tolerance = Scalar(2e-16);
482 const Scalar alfbet = alpha + beta;
483 const Scalar softLo = lo + Scalar(1e-15);
484 const Scalar softHi = hi - Scalar(1e-15);
485 if (!(softHi > softLo)) {
486 return false;
487 }
488 if (z <= softLo) {
489 z = softLo + Scalar(0.1) * (softHi - softLo);
490 }
491 if (z >= softHi) {
492 z = softHi - Scalar(0.1) * (softHi - softLo);
493 }
494
495 Scalar a, b, c, p1, p2, p3, pp, temp, z1;
496 std::size_t its = 1;
497 do {
498 temp = 2.0 + alfbet;
499 p1 = (alpha - beta + temp * z) / 2.0;
500 p2 = 1.0;
501 for (std::size_t j = 2; j <= n; ++j) {
502 p3 = p2;
503 p2 = p1;
504 temp = 2 * j + alfbet;
505 a = 2 * j * (j + alfbet) * (temp - 2.0);
506 b = (temp - 1.0) * (alpha * alpha - beta * beta + temp * (temp - 2.0) * z);
507 c = 2.0 * (j - 1 + alpha) * (j - 1 + beta) * temp;
508 p1 = (b * p2 - c * p3) / a;
509 }
510 const Scalar one_m_z2 = Scalar(1) - z * z;
511 if (Kokkos::abs(one_m_z2) < Scalar(1e-30)) {
512 return false;
513 }
514 pp = (n * (alpha - beta - temp * z) * p1 + 2.0 * (n + alpha) * (n + beta) * p2)
515 / (temp * one_m_z2);
516 z1 = z;
517 z = z1 - p1 / pp;
518 if (z < softLo) {
519 z = softLo;
520 }
521 if (z > softHi) {
522 z = softHi;
523 }
524 if (its > minNewtonIterations && Kokkos::abs(z - z1) <= tolerance
525 && Kokkos::abs(p1) <= Scalar(1e-12)) {
526 p_out = p1;
527 dp_out = pp;
528 pnm1_out = p2;
529 temp_out = temp;
530 return true;
531 }
532 ++its;
533 } while (its <= maxNewtonIterations);
534 p_out = p1;
535 dp_out = pp;
536 pnm1_out = p2;
537 temp_out = temp;
538 return false;
539 }

545 template <typename Scalar>
546 Scalar brentJacobiRoot(std::size_t n, Scalar alpha, Scalar beta, Scalar lo, Scalar hi,
547 std::size_t maxIter = 200) {
548 Scalar a = lo;
549 Scalar b = hi;
550 Scalar fa, dfa, fb, dfb;
551 evalJacobiAndDeriv(n, alpha, beta, a, fa, dfa);
552 evalJacobiAndDeriv(n, alpha, beta, b, fb, dfb);
553 if (fa * fb > Scalar(0)) {
554 throw std::runtime_error("nodes1d::brentJacobiRoot: bracket does not change sign");
555 }
556 if (Kokkos::abs(fa) < Kokkos::abs(fb)) {
557 std::swap(a, b);
558 std::swap(fa, fb);
559 }
560 Scalar c = a;
561 Scalar fc = fa;
562 bool mflag = true;
563 Scalar d = Scalar(0);
564 for (std::size_t it = 0; it < maxIter; ++it) {
565 if (fb == Scalar(0) || Kokkos::abs(b - a) < Scalar(2e-16)) {
566 return b;
567 }
568 Scalar s;
569 if (fa != fc && fb != fc) {
570 s = a * fb * fc / ((fa - fb) * (fa - fc)) + b * fa * fc / ((fb - fa) * (fb - fc))
571 + c * fa * fb / ((fc - fa) * (fc - fb));
572 } else {
573 s = b - fb * (b - a) / (fb - fa);
574 }
575 const bool cond1 = (s < (Scalar(3) * a + b) / Scalar(4) || s > b);
576 const bool cond2 = mflag && Kokkos::abs(s - b) >= Kokkos::abs(b - c) / Scalar(2);
577 const bool cond3 = !mflag && Kokkos::abs(s - b) >= Kokkos::abs(c - d) / Scalar(2);
578 const bool cond4 = mflag && Kokkos::abs(b - c) < Scalar(2e-16);
579 const bool cond5 = !mflag && Kokkos::abs(c - d) < Scalar(2e-16);
580 if (cond1 || cond2 || cond3 || cond4 || cond5) {
581 s = (a + b) / Scalar(2);
582 mflag = true;
583 } else {
584 mflag = false;
585 }
586 Scalar fs, dfs;
587 evalJacobiAndDeriv(n, alpha, beta, s, fs, dfs);
588 d = c;
589 c = b;
590 fc = fb;
591 if (fa * fs < Scalar(0)) {
592 b = s;
593 fb = fs;
594 } else {
595 a = s;
596 fa = fs;
597 }
598 if (Kokkos::abs(fa) < Kokkos::abs(fb)) {
599 std::swap(a, b);
600 std::swap(fa, fb);
601 }
602 }
603 return b;
604 }

611 bool findSignChange(std::size_t n, Scalar alpha, Scalar beta, Scalar lo, Scalar hi, Scalar& a,
612 Scalar& b, int samples = 128) {
613 Scalar fa, dfa;
614 evalJacobiAndDeriv(n, alpha, beta, lo, fa, dfa);
615 Scalar prev = lo;
616 Scalar fprev = fa;
617 for (int s = 1; s <= samples; ++s) {

634 template <typename Scalar>

707 template <typename Scalar>
708 void collectJacobiRootsBySignChange(std::size_t n, Scalar alpha, Scalar beta,
709 std::vector<Scalar>& roots) {
710 roots.clear();
711 const Scalar glo = Scalar(-1) + Scalar(1e-14);
712 const Scalar ghi = Scalar(1) - Scalar(1e-14);
713 const std::size_t samples = jacobiBrentSampleCount(n);
714 // x = -cos(theta): uniform in theta matches endpoint clustering of Jacobi zeros.
715 const Scalar t0 = Kokkos::acos(-glo);
716 const Scalar t1 = Kokkos::acos(-ghi);
717 Scalar prev = glo;
718 Scalar fprev, df;
719 evalJacobiAndDeriv(n, alpha, beta, prev, fprev, df);
720 for (std::size_t s = 1; s <= samples; ++s) {
721 const Scalar theta =
722 t0 + (t1 - t0) * (static_cast<Scalar>(s) / static_cast<Scalar>(samples));
723 const Scalar x = -Kokkos::cos(theta);
724 Scalar fx;
725 evalJacobiAndDeriv(n, alpha, beta, x, fx, df);
726 if (fprev * fx <= Scalar(0) && roots.size() < n) {
727 Scalar a = prev;
728 Scalar b = x;
729 if (b > a) {
730 roots.push_back(brentJacobiRoot(n, alpha, beta, a, b));
731 }
732 }
733 prev = x;
734 fprev = fx;
735 }
736 }

749 template <typename Scalar>
750 void computeGaussJacobiNewton(std::size_t n, Scalar alpha, Scalar beta, Scalar* nodes,
751 Scalar* weights, std::size_t maxNewtonIterations,
752 std::size_t minNewtonIterations, InitialGuessType initialGuess) {
753 const Scalar sepTol = Scalar(1e-13);
754 const Scalar alfbet = alpha + beta;
755 const Scalar mu0 =
756 Kokkos::exp(Kokkos::lgamma(alpha + n) + Kokkos::lgamma(beta + n)
757 - Kokkos::lgamma(static_cast<Scalar>(n) + 1.)
758 - Kokkos::lgamma(static_cast<Scalar>(n) + alfbet + 1.0))
759 * Kokkos::pow(Scalar(2), alfbet);
760 const Scalar glo = Scalar(-1) + Scalar(1e-14);
761 const Scalar ghi = Scalar(1) - Scalar(1e-14);
762
763 const std::vector<InitialGuessType> ladder = buildGuessLadder(initialGuess);
764 std::vector<Scalar> unique;
765
766 for (InitialGuessType kind : ladder) {
767 for (std::size_t k = 0; k < n; ++k) {
768 Scalar z;
769 if (!tryNewtonJacobiRoot(k, kind, n, alpha, beta, unique, maxNewtonIterations,
770 minNewtonIterations, glo, ghi, z)) {
771 continue;
772 }
773 unique.push_back(z);
774 }
775 unique = dedupeSorted(std::move(unique), sepTol);
776 if (unique.size() == n) {
777 break;
778 }
779 }
780
781 if (unique.size() != n) {
782 collectJacobiRootsBySignChange(n, alpha, beta, unique);
783 unique = dedupeSorted(std::move(unique), sepTol);
784 }
785
786 if (unique.size() != n) {
787 throw std::runtime_error(
788 "nodes1d::computeGaussJacobi (Newton): expected " + std::to_string(n)
789 + " distinct roots, found " + std::to_string(unique.size()));
790 }
791
792 for (std::size_t i = 0; i < n; ++i) {
793 nodes[i] = unique[i];
794 Scalar z = nodes[i], p1, pp, p2, temp;
795 newtonOneRootBounded(n, alpha, beta, z, glo, ghi, maxNewtonIterations,
796 minNewtonIterations, p1, pp, p2, temp);
797 weights[i] = mu0 * temp / (pp * p2);
798 }
799 assertDistinctSortedNodes("nodes1d::computeGaussJacobi (Newton)", nodes, n, sepTol);
800
801 if (alpha == beta) {
802 for (std::size_t i = 0; i < n; ++i) {
803 const std::size_t j = n - 1 - i;
804 if (i > j) {
805 break;
806 }
807 const Scalar xsym = Scalar(0.5) * (nodes[i] - nodes[j]);
808 const Scalar wsym = Scalar(0.5) * (weights[i] + weights[j]);
809 nodes[i] = xsym;
810 nodes[j] = -xsym;
811 weights[i] = weights[j] = wsym;
812 }
813 }
814 }
