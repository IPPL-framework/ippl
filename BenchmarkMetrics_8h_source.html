38 inline double computeMean(const std::vector<double>& values) {
39 if (values.empty()) {
40 return 0.0;
41 }
42 return std::accumulate(values.begin(), values.end(), 0.0) / values.size();
43 }

67 template <std::size_t N>
68 void writeBMF(const std::string& filename,
69 const std::array<std::string_view, N>& whitelist) {
70 if (ippl::Comm->rank() != 0) {
71 return;
72 }
73
74 std::vector<std::string> timerNames = IpplTimings::getTimerNames();
75
76 std::ostringstream json;
77 json << std::setprecision(17) << std::scientific;
78 json << "{\n";
79
80 bool first = true;
81 for (const auto& name : timerNames) {
82 if (!detail::isWhitelisted(name, whitelist)) {
83 continue;
84 }
85
86 IpplTimings::TimerRef ref = IpplTimings::getTimer(name.c_str());
87 std::vector<double> values = IpplTimings::getMeasurements(ref);
88 std::size_t count = values.size();
89
90 if (count == 0) {
91 continue;
92 }
93
94 // IpplTimings stores values in seconds. Bencher's latency measure
95 // uses nanoseconds, so scale before emitting.
96 constexpr double seconds_to_nanoseconds = 1.0e9;
97 double mean = detail::computeMean(values) * seconds_to_nanoseconds;
98 double minv = detail::computeMin(values) * seconds_to_nanoseconds;
99 double maxv = detail::computeMax(values) * seconds_to_nanoseconds;
100
101 if (!first) {
102 json << ",\n";
103 }
104 first = false;
105
106 json << " \"" << name << "\": {\n";
107 json << " \"latency\": {\n";
108 json << " \"value\": " << mean << ",\n";
109 json << " \"lower_value\": " << minv << ",\n";
110 json << " \"upper_value\": " << maxv << "\n";
111 json << " },\n";
112 json << " \"count\": {\n";
113 json << " \"value\": " << count << "\n";
114 json << " }\n";
115 json << " }";
116 }
117
118 json << "\n}\n";
119
120 std::ofstream out(filename);
121 out << json.str();
122 }
