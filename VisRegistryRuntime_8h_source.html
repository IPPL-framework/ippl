123 template <typename VisitorT>
124 bool forOne(const std::string& label, VisitorT& v) const {
125 auto it = indexExec_m.find(label);
126 if (it == indexExec_m.end()) {
127 std::cerr << "VisRegistryRuntime::forOne: label not found: '" << label << "'\n";
128 std::cerr << " Available registry labels (" << indexExec_m.size() << "):" << std::endl;
129 for (const auto& kv : indexExec_m) std::cerr << " " << kv.first << std::endl;
130 std::cerr << std::endl;
131 return false;
132 }
133 const auto& e = entries_m[it->second];
134
135 if constexpr (std::is_constructible_v<CatalystAdaptor::VisVisitorVariant_t, VisitorT*>) {
136 if (!e.dispatchVis_m) {
137 std::cout << "forOne: No vis dispatch CallBack" << std::endl;
138 return false;
139 }
140 CatalystAdaptor::VisVisitorVariant_t var = &v;
141 e.dispatchVis_m(var);
142 return true;
143 } else if constexpr (std::is_constructible_v<CatalystAdaptor::SteerVisitorVariant_t, VisitorT*>) {
144 if (!e.dispatchSteer_m) {
145 std::cout << "forOne: No steer dispatch CallBack" << std::endl;
146 return false;
147 }
148 CatalystAdaptor::SteerVisitorVariant_t var = &v;
149 e.dispatchSteer_m(var);
150 return true;
151 } else {
152 return false;
153 }
