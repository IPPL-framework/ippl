13 template <typename Attrib>
14 bool OrthogonalRecursiveBisection<Field, Tp>::binaryRepartition(
15 const Attrib& R, FieldLayout<Dim>& fl, const bool& isFirstRepartition,
16 const std::array<bool, Dim>& forbiddenAxes) {
17 // Timings
18 static IpplTimings::TimerRef tbasicOp = IpplTimings::getTimer("basicOperations");
19 static IpplTimings::TimerRef tperpReduction = IpplTimings::getTimer("perpReduction");
20 static IpplTimings::TimerRef tallReduce = IpplTimings::getTimer("allReduce");
21 static IpplTimings::TimerRef tscatter = IpplTimings::getTimer("scatterR");
22
23 // Scattering of particle positions in field
24 // In case of first repartition we know the density from the
25 // analytical expression and we use that for load balancing
26 // and create particles. Note the particles are created only
27 // after the first repartition and hence we cannot call scatter
28 // before it.
29 IpplTimings::startTimer(tscatter);
30 if (!isFirstRepartition) {
31 scatterR(R);
32 }
33
34 IpplTimings::stopTimer(tscatter);
35
36 IpplTimings::startTimer(tbasicOp);
37
38 // Get number of ranks
39 auto& comm = fl.comm;
40 int nprocs = comm.size();
41
42 // Start with whole domain and total number of nodes
43 std::vector<NDIndex<Dim>> domains = {fl.getDomain()};
44 std::vector<int> procs = {nprocs};
45
46 // Arrays for reduction
47 std::vector<Tf> reduced, reducedRank;
48
49 // Start recursive repartition loop
50 unsigned int it = 0;
51 int maxprocs = nprocs;
52 IpplTimings::stopTimer(tbasicOp);
53
54 while (maxprocs > 1) {
55 // Find cut axis
56 IpplTimings::startTimer(tbasicOp);
57 int cutAxis = findCutAxis(domains[it], forbiddenAxes);
58 if (cutAxis < 0) {
59 IpplTimings::stopTimer(tbasicOp);
60 return false;
61 }
62 int cutAxisLength = static_cast<int>(domains[it][cutAxis].length());
63 if (cutAxisLength < 4) {
64 IpplTimings::stopTimer(tbasicOp);
65 return false;
66 }
67 IpplTimings::stopTimer(tbasicOp);
68
69 // Reserve space
70 IpplTimings::startTimer(tperpReduction);
71 reduced.resize(cutAxisLength);
72 reducedRank.resize(cutAxisLength);
73
74 std::fill(reducedRank.begin(), reducedRank.end(), 0.0);
75 std::fill(reduced.begin(), reduced.end(), 0.0);
76
77 // Peform reduction with field of weights and communicate to the other ranks
78 perpendicularReduction(reducedRank, cutAxis, domains[it]);
79 IpplTimings::stopTimer(tperpReduction);
80
81 // Communicate to all the reduced weights
82 IpplTimings::startTimer(tallReduce);
83 comm.allreduce(reducedRank.data(), reduced.data(), reducedRank.size(), std::plus<Tf>());
84 IpplTimings::stopTimer(tallReduce);
85
86 // Find median of reduced weights
87 IpplTimings::startTimer(tbasicOp);
88 // Initialize median to some value (1 is lower bound value)
89 int median = 1;
90 median = std::clamp(findMedian(reduced), 1, cutAxisLength - 3);
91 IpplTimings::stopTimer(tbasicOp);
92
93 // Cut domains and procs
94 IpplTimings::startTimer(tbasicOp);
95 cutDomain(domains, procs, it, cutAxis, median);
96
97 // Update max procs
98 maxprocs = 0;
99 for (unsigned int i = 0; i < procs.size(); i++) {
100 if (procs[i] > maxprocs) {
101 maxprocs = procs[i];
102 it = i;
103 }
104 }
105 IpplTimings::stopTimer(tbasicOp);
106
107 // Clear all arrays
108 IpplTimings::startTimer(tperpReduction);
109 reduced.clear();
110 reducedRank.clear();
111 IpplTimings::stopTimer(tperpReduction);
112 }
113
114 IpplTimings::startTimer(tbasicOp);
115 const NDIndex<Dim>& globalDomain = fl.getDomain();
116 if (domains.size() != static_cast<size_t>(nprocs)) {
117 IpplTimings::stopTimer(tbasicOp);
118 return false;
119 }
120 if (!domainsTileAllowedDecomposition(domains, globalDomain, forbiddenAxes)) {
121 IpplTimings::stopTimer(tbasicOp);
122 return false;
123 }
124
125 // Update FieldLayout with new indices
126 fl.updateLayout(domains);
127
128 // Update local field with new layout
129 bf_m.updateLayout(fl);
130 IpplTimings::stopTimer(tbasicOp);
131
132 return true;
133 }

136 int OrthogonalRecursiveBisection<Field, Tp>::findCutAxis(
137 const NDIndex<Dim>& dom, const std::array<bool, Dim>& forbiddenAxes) {
138 int cutAxis = -1;
139 for (unsigned d = 0; d < Dim; ++d) {
140 if (forbiddenAxes[d]) {
141 continue;
142 }
143 if (cutAxis < 0 || dom[d].length() > dom[cutAxis].length()) {
144 cutAxis = d;
145 }
146 }

161 template <class Field, class Tp>
162 bool OrthogonalRecursiveBisection<Field, Tp>::domainsTileAllowedDecomposition(
163 const std::vector<NDIndex<Dim>>& domains, const NDIndex<Dim>& globalDomain,
164 const std::array<bool, Dim>& forbiddenAxes) const {
165 size_t globalCells = 1;
166 for (unsigned d = 0; d < Dim; ++d) {
167 globalCells *= globalDomain[d].length();
168 }
169
170 size_t localCells = 0;
171 for (size_t i = 0; i < domains.size(); ++i) {
172 const auto& domain = domains[i];
173 size_t domainCells = 1;
174 for (unsigned d = 0; d < Dim; ++d) {
175 if (domain[d].length() < 2) {
176 return false;
177 }
178 if (domain[d].first() < globalDomain[d].first()
179 || domain[d].last() > globalDomain[d].last()) {
180 return false;
181 }
182 if (forbiddenAxes[d] && domain[d] != globalDomain[d]) {
183 return false;
184 }
185 domainCells *= domain[d].length();
186 }
187
188 for (size_t j = 0; j < i; ++j) {
189 if (domainsOverlap(domain, domains[j])) {
190 return false;
191 }
192 }
193 localCells += domainCells;
194 }
195
196 return localCells == globalCells;
197 }

199 template <class Field, class Tp>
200 void OrthogonalRecursiveBisection<Field, Tp>::perpendicularReduction(
201 std::vector<Tf>& rankWeights, unsigned int cutAxis, NDIndex<Dim>& dom) {
202 // Check if domains overlap, if not no need for reduction
203 NDIndex<Dim> lDom = bf_m.getOwned();
204 if (lDom[cutAxis].first() > dom[cutAxis].last()
205 || lDom[cutAxis].last() < dom[cutAxis].first()) {
206 return;
207 }
208
209 // Get field's local weights
210 int nghost = bf_m.getNghost();
211 const auto data = bf_m.getView();
212
213 // Determine the iteration bounds of the reduction
214 int cutAxisFirst =
215 std::max(lDom[cutAxis].first(), dom[cutAxis].first()) - lDom[cutAxis].first() + nghost;
216 int cutAxisLast =
217 std::min(lDom[cutAxis].last(), dom[cutAxis].last()) - lDom[cutAxis].first() + nghost;
218
219 // Set iterator for where to write in the reduced array
220 unsigned int arrayStart = 0;
221 if (dom[cutAxis].first() < lDom[cutAxis].first()) {
222 arrayStart = lDom[cutAxis].first() - dom[cutAxis].first();
223 }
224
225 // Find all the perpendicular axes
226 using exec_space = typename Field::execution_space;
227 using index_type = typename RangePolicy<Dim, exec_space>::index_type;
228 Kokkos::Array<index_type, Dim> begin, end;
229 for (unsigned d = 0; d < Dim; d++) {
230 if (d == cutAxis) {
231 continue;
232 }
233
234 int inf = std::max(lDom[d].first(), dom[d].first()) - lDom[d].first() + nghost;
235 int sup = std::min(lDom[d].last(), dom[d].last()) - lDom[d].first() + nghost;
236 // inf and sup bounds must be within the domain to reduce, if not no need to reduce
237 if (sup < inf) {
238 return;
239 }
240
241 begin[d] = inf;
242 // The +1 is for Kokkos loop
243 end[d] = sup + 1;
244 }
245
246 // Iterate along cutAxis
247 for (int i = cutAxisFirst; i <= cutAxisLast; i++) {
248 begin[cutAxis] = i;
249 end[cutAxis] = i + 1;
250
251 // Reducing over perpendicular plane defined by cutAxis
252 Tf tempRes = Tf(0);
253
254 using index_array_type = typename RangePolicy<Dim, exec_space>::index_array_type;
255 ippl::parallel_reduce(
256 "ORB weight reduction", createRangePolicy<Dim, exec_space>(begin, end),
257 KOKKOS_LAMBDA(const index_array_type& args, Tf& weight) {
258 weight += apply(data, args);
259 },
260 Kokkos::Sum<Tf>(tempRes));
261
262 Kokkos::fence();
263
264 rankWeights[arrayStart++] = tempRes;
265 }
266 }

268 template <class Field, class Tp>
269 int OrthogonalRecursiveBisection<Field, Tp>::findMedian(std::vector<Tf>& w) {
270 // Special case when array must be cut in half in order to not have planes
271 if (w.size() == 4) {
272 return 1;
273 }
274 // Get total sum of array
275 Tf tot = std::accumulate(w.begin(), w.end(), Tf(0));
276
277 // Find position of median as half of total in array
278 Tf half = 0.5 * tot;
279 Tf curr = Tf(0);
280 // Do not need to iterate to full extent since it must not give planes
281 for (unsigned int i = 0; i < w.size() - 1; i++) {
282 curr += w[i];
283 if (curr >= half) {
284 // If all particles are in the first plane, cut at 1 so to have size 2
285 if (i == 0) {
286 return 1;
287 }
288 Tf previous = curr - w[i];
289 // curr - half < half - previous
290 if ((curr + previous) <= tot
291 && curr != half) { // if true then take current i, otherwise i-1
292 if (i == w.size() - 2) {
293 return (i - 1);
294 } else {
295 return i;
296 }
297 } else {
298 return (i > 1) ? (i - 1) : 1;
299 }
300 }
301 }
302 // If all particles are in the last plane, cut two indices before the end so to have size 2
303 return w.size() - 3;
304 }

306 template <class Field, class Tp>
307 void OrthogonalRecursiveBisection<Field, Tp>::cutDomain(std::vector<NDIndex<Dim>>& domains,
308 std::vector<int>& procs, int it,
309 int cutAxis, int median) {
310 // Cut domains[it] in half at median along cutAxis
311 NDIndex<Dim> leftDom, rightDom;
312 domains[it].split(leftDom, rightDom, cutAxis, median + domains[it][cutAxis].first());
313 domains[it] = leftDom;
314 domains.insert(domains.begin() + it + 1, rightDom);
315
316 // Cut procs in half
317 int temp = procs[it];
318 procs[it] = procs[it] / 2;
319 procs.insert(procs.begin() + it + 1, temp - procs[it]);
320 }

324 void OrthogonalRecursiveBisection<Field, Tp>::scatterR(const Attrib& r) {
325 using vector_type = typename mesh_type::vector_type;
326 static_assert(
327 Kokkos::SpaceAccessibility<typename Attrib::memory_space,
328 typename Field::memory_space>::accessible,
329 "Particle attribute memory space must be accessible from ORB field memory space");
330
331 // Reset local field
332 bf_m = 0.0;
333 // Get local data
334 auto view = bf_m.getView();
335 const mesh_type& mesh = bf_m.get_mesh();
336 const FieldLayout<Dim>& layout = bf_m.getLayout();
337 const NDIndex<Dim>& lDom = layout.getLocalNDIndex();
338 const int nghost = bf_m.getNghost();
339
340 // Get spacings
341 const vector_type& dx = mesh.getMeshSpacing();
342 const vector_type& origin = mesh.getOrigin();
343 const vector_type invdx = 1.0 / dx;
344
345 using policy_type = Kokkos::RangePolicy<size_t, typename Field::execution_space>;
346
347 // Extract the view to avoid capturing the entire ParticleAttrib in the lambda
348 // (ParticleAttrib's destructor is host-only)
349 auto rview = r.getView();
350
351 Kokkos::parallel_for(
352 "ParticleAttrib::scatterR", policy_type(0, r.getParticleCount()),
353 KOKKOS_LAMBDA(const size_t idx) {
354 // Find nearest grid point
355 Vector<Tp, Dim> l = (rview(idx) - origin) * invdx + 0.5;
356 Vector<int, Dim> index = l;
357 Vector<Tf, Dim> whi = l - index;
358 Vector<Tf, Dim> wlo = 1.0 - whi;
359
360 Vector<size_t, Dim> args = index - lDom.first() + nghost;
361
362 // Scatter
363 scatterToField(std::make_index_sequence<1 << Dim>{}, view, wlo, whi, args);
364 });
365
366 bf_m.accumulateHalo();
367 }
