27 Quadrature<T, NumNodes1D, ElementType>::getWeightsForRefElement() const {

30 Vector<T, std::remove_reference_t<decltype(*this)>::numElementNodes> tensor_prod_w;
31
32 Vector<unsigned, ElementType::dim> nd_index(0);
33 for (unsigned i = 0; i < std::remove_reference_t<decltype(*this)>::numElementNodes; ++i) {
34 tensor_prod_w[i] = 1.0;
35 for (unsigned d = 0; d < ElementType::dim; ++d) {
36 tensor_prod_w[i] *= w[nd_index[d]];
37 }
38
39 // Update nd_index for next iteration

44 break;
45 nd_index[d] = 0;
46 }
47 }
